function mauPlotKey(l: location) :: text:
    if {mauplot::world} is not set:
        return "OUTSIDE"
    if world of {_l} is not {mauplot::world}:
        return "OUTSIDE"

    if {mauplot::grid-enabled} is true:
        set {_blockX} to floor(x-coordinate of {_l})
        set {_blockZ} to floor(z-coordinate of {_l})
        set {_relX} to {_blockX} - {mauplot::originX}
        set {_relZ} to {_blockZ} - {mauplot::originZ}
        set {_step} to {mauplot::grid-step}

        if {_relX} < 0:
            return "OUTSIDE"
        if {_relZ} < 0:
            return "OUTSIDE"
        set {_gridLimit} to {mauplot::grid-count} * {_step}
        if {_relX} >= {_gridLimit}:
            return "OUTSIDE"
        if {_relZ} >= {_gridLimit}:
            return "OUTSIDE"

        set {_plotX} to floor({_relX} / {_step})
        set {_plotZ} to floor({_relZ} / {_step})
        set {_insideX} to {_relX} - ({_plotX} * {_step})
        set {_insideZ} to {_relZ} - ({_plotZ} * {_step})
        set {_worldName} to world of {_l}
        set {_cellKey} to "%{_worldName}%::%{_plotX}%::%{_plotZ}%"

        # Normale Plotfläche.
        if {_insideX} < {@plot-size}:
            if {_insideZ} < {@plot-size}:
                return {_cellKey}

        # Die 12 Blöcke breite Oststraße gehört bei /p merge zum westlichen Plot.
        if {_insideX} >= {@plot-size}:
            if {_insideZ} < {@plot-size}:
                set {_neighborX} to {_plotX} + 1
                if {_neighborX} < {mauplot::grid-count}:
                    set {_neighborKey} to "%{_worldName}%::%{_neighborX}%::%{_plotZ}%"
                    if {mauplot::merge::%{_cellKey}%::%{_neighborKey}%} is true:
                        return {_cellKey}
            return "OUTSIDE"

        # Die 12 Blöcke breite Südstraße gehört bei /p merge zum nördlichen Plot.
        if {_insideZ} >= {@plot-size}:
            if {_insideX} < {@plot-size}:
                set {_neighborZ} to {_plotZ} + 1
                if {_neighborZ} < {mauplot::grid-count}:
                    set {_neighborKey} to "%{_worldName}%::%{_plotX}%::%{_neighborZ}%"
                    if {mauplot::merge::%{_cellKey}%::%{_neighborKey}%} is true:
                        return {_cellKey}
            return "OUTSIDE"

        # Straßenkreuzungen bleiben geschützt, sofern sie nicht Teil einer geraden Merge-Verbindung sind.
        return "OUTSIDE"

    # Rückwärtskompatibilität für eine manuell eingerichtete Plotwelt ohne 10K-Raster.
    set {_plotX} to floor((x-coordinate of {_l}) / {@plot-size})
    set {_plotZ} to floor((z-coordinate of {_l}) / {@plot-size})
    set {_worldName} to world of {_l}
    return "%{_worldName}%::%{_plotX}%::%{_plotZ}%"

function mauPlotGeneratedRoad(l: location) :: boolean:
    if {mauplot::grid-enabled} is not true:
        return false
    if {mauplot::world} is not set:
        return false
    if world of {_l} is not {mauplot::world}:
        return false
    set {_blockX} to floor(x-coordinate of {_l})
    set {_blockZ} to floor(z-coordinate of {_l})
    set {_relX} to {_blockX} - {mauplot::originX}
    set {_relZ} to {_blockZ} - {mauplot::originZ}
    set {_gridLimit} to {mauplot::grid-count} * {mauplot::grid-step}
    if {_relX} < 0:
        return false
    if {_relZ} < 0:
        return false
    if {_relX} >= {_gridLimit}:
        return false
    if {_relZ} >= {_gridLimit}:
        return false
    if mauPlotKey({_l}) is "OUTSIDE":
        return true
    return false

function mauPlotCount(p: player) :: number:
    set {_uuid} to "%uuid of {_p}%"
    # Migration des alten Ein-Plot-Formats in die neue Plotliste.
    if {mauplot::playerplot::%{_uuid}%} is set:
        set {_oldKey} to {mauplot::playerplot::%{_uuid}%}
        if {mauplot::playerplots::%{_uuid}%::*} does not contain {_oldKey}:
            add {_oldKey} to {mauplot::playerplots::%{_uuid}%::*}
        if {mauplot::home::%{_oldKey}%} is not set:
            if {mauplot::home::%{_uuid}%} is set:
                set {mauplot::home::%{_oldKey}%} to {mauplot::home::%{_uuid}%}
    set {_count} to size of {mauplot::playerplots::%{_uuid}%::*}
    return {_count}

function mauPlotMax(p: player) :: number:
    set {_uuid} to "%uuid of {_p}%"
    set {_extra} to {mauplot::extra-slots::%{_uuid}%}
    if {_extra} is not set:
        set {_extra} to 0
    set {_maximum} to {@plot-max-per-player} + {_extra}
    return {_maximum}

function mauPlotIsAdmin(p: player) :: boolean:
    # OPs erhalten den Admin-Bypass auch ohne extra Permission-Plugin.
    if {_p} is op:
        return true
    if {_p} has permission "{@plot-admin-permission}":
        return true
    if {_p} has permission "{@mau-world-admin-permission}":
        return true
    return false

function mauPlotCanBuild(p: player, k: text) :: boolean:
    if mauPlotIsAdmin({_p}) is true:
        return true
    set {_uuid} to "%uuid of {_p}%"
    set {_owner} to {mauplot::owner::%{_k}%}
    if {_owner} is not set:
        return false
    if {_owner} is {_uuid}:
        return true
    if {mauplot::denied::%{_k}%::%{_uuid}%} is set:
        return false
    # TRUST bleibt dauerhaft aktiv. ADD gilt nur, solange der Besitzer online ist.
    if {mauplot::trusted::%{_k}%::%{_uuid}%} is set:
        return true
    if {mauplot::added::%{_k}%::%{_uuid}%} is set:
        loop all players:
            if "%uuid of loop-player%" is {_owner}:
                return true
    return false

function mauPlotCellAvailable(plotX: number, plotZ: number, w: world) :: boolean:
    if {mauplot::grid-enabled} is true:
        set {_lastIndex} to {mauplot::grid-count} - 1
        # Getrennte Grenzprüfungen sind mit Skript 2.17.1 eindeutig parsebar.
        if {_plotX} < 0:
            return false
        if {_plotX} > {_lastIndex}:
            return false
        if {_plotZ} < 0:
            return false
        if {_plotZ} > {_lastIndex}:
            return false
    set {_key} to "%{_w}%::%{_plotX}%::%{_plotZ}%"
    if {mauplot::owner::%{_key}%} is set:
        return false
    if {mauplot::locked::%{_key}%} is set:
        return false
    # Kein Spawn-Abstand: Auch das Plot direkt am Spawn darf beansprucht werden.
    return true

function mauPlotHomeLocation(plotX: number, plotZ: number, w: world) :: location:
    if {mauplot::grid-enabled} is true:
        set {_centerX} to {mauplot::originX} + ({_plotX} * {mauplot::grid-step}) + ({@plot-size} / 2)
        set {_centerZ} to {mauplot::originZ} + ({_plotZ} * {mauplot::grid-step}) + ({@plot-size} / 2)
    else:
        set {_centerX} to ({_plotX} * {@plot-size}) + ({@plot-size} / 2)
        set {_centerZ} to ({_plotZ} * {@plot-size}) + ({@plot-size} / 2)
    if {mauplot::grid-enabled} is true:
        # Alle generierten MAU-Plots haben ihre ebene Oberseite auf Y=280.
        set {_loc} to location at {_centerX}, 281, {_centerZ} of {_w}
        return {_loc}
    set {_loc} to location at {_centerX}, 0, {_centerZ} of {_w}
    set {_ground} to highest solid block at {_loc}
    set y-coordinate of {_loc} to (y-coordinate of {_ground}) + 1
    return {_loc}

function mauPlotGenerateCell(plotX: number, plotZ: number, w: world):
    set {_lastIndex} to {mauplot::grid-count} - 1
    if {_plotX} >= 0:
        if {_plotZ} >= 0:
            if {_plotX} <= {_lastIndex}:
                if {_plotZ} <= {_lastIndex}:
                    set {_key} to "%{_w}%::%{_plotX}%::%{_plotZ}%"
                    if {mauplot::generated::%{_key}%} is not set:
                        # Die zentrale 6x6-Startstadt existiert bereits und wird nicht neu gebaut.
                        set {_isStartCityCell} to false
                        if {_plotX} >= 47:
                            if {_plotX} <= 52:
                                if {_plotZ} >= 47:
                                    if {_plotZ} <= 52:
                                        set {_isStartCityCell} to true
                        if {_isStartCityCell} is true:
                            set {mauplot::generated::%{_key}%} to true
                        else:
                            mauPlotBuildCell({_plotX}, {_plotZ}, {_w}, {_key})

                    # Auch vorhandene Startstadt-Zellen übernehmen die globale bzw. lokale Tiefe.
                    set {_requiredDepth} to mauPlotBedrockDepth({_key})
                    if {_requiredDepth} > 0:
                        set {mauplot::bedrock-enabled} to true
                        set {_x1} to {mauplot::originX} + ({_plotX} * {mauplot::grid-step})
                        set {_z1} to {mauplot::originZ} + ({_plotZ} * {mauplot::grid-step})
                        set {_x2} to {_x1} + 63
                        set {_z2} to {_z1} + 63
                        mauPlotQueueBedrockRegion({_key}, {_w}, {_x1}, {_z1}, {_x2}, {_z2}, {_requiredDepth})
                    mauPlotQueueRoadSupport({_plotX}, {_plotZ}, {_w}, {_key})

function mauPlotQueueRoadWall(id: text, w: world, axis: text, fixed: number, start: number):
    if {mauplot::roadwall::head} is not set:
        set {mauplot::roadwall::head} to 1
    if {mauplot::roadwall::tail} is not set:
        set {mauplot::roadwall::tail} to 0
    add 1 to {mauplot::roadwall::tail}
    set {_index} to {mauplot::roadwall::tail}
    set {mauplot::roadwall::queue::%{_index}%::id} to {_id}
    set {mauplot::roadwall::queue::%{_index}%::world} to {_w}
    set {mauplot::roadwall::queue::%{_index}%::axis} to {_axis}
    set {mauplot::roadwall::queue::%{_index}%::fixed} to {_fixed}
    set {mauplot::roadwall::queue::%{_index}%::start} to {_start}
    set {mauplot::roadwall::queue::%{_index}%::tile} to 0

function mauPlotQueueRoadSupport(plotX: number, plotZ: number, w: world, plotKey: text):
    if {mauplot::road-support-queued::%{_plotKey}%} is true:
        stop
    set {mauplot::road-support-queued::%{_plotKey}%} to true
    set {_lastIndex} to {mauplot::grid-count} - 1
    set {_x} to {mauplot::originX} + ({_plotX} * {mauplot::grid-step})
    set {_z} to {mauplot::originZ} + ({_plotZ} * {mauplot::grid-step})
    if {_plotX} < {_lastIndex}:
        mauPlotQueueRoadWall("%{_plotKey}%::east-left", {_w}, "x", ({_x} + 67), {_z})
        mauPlotQueueRoadWall("%{_plotKey}%::east-right", {_w}, "x", ({_x} + 73), {_z})
    if {_plotZ} < {_lastIndex}:
        mauPlotQueueRoadWall("%{_plotKey}%::south-left", {_w}, "z", ({_z} + 67), {_x})
        mauPlotQueueRoadWall("%{_plotKey}%::south-right", {_w}, "z", ({_z} + 73), {_x})

        set all blocks within location({_rx1}, 279, {_z}, {_w}) and location({_rx2}, 279, {_rz2}, {_w}) to bedrock

        set {_bz2} to {_z} + 75

        set all blocks within location({_x}, 280, {_bz1}, {_w}) and location({_bx2}, 280, {_bz2}, {_w}) to gray_concrete
        set all blocks within location({_x}, 280, ({_z} + 64), {_w}) and location({_bx2}, 280, ({_z} + 66), {_w}) to smooth_stone
        set all blocks within location({_x}, 280, ({_z} + 67), {_w}) and location({_bx2}, 280, ({_z} + 67), {_w}) to stone_bricks
        set all blocks within location({_x}, 280, ({_z} + 70), {_w}) and location({_bx2}, 280, ({_z} + 70), {_w}) to yellow_concrete
        set all blocks within location({_x}, 280, ({_z} + 73), {_w}) and location({_bx2}, 280, ({_z} + 73), {_w}) to stone_bricks
        set all blocks within location({_x}, 280, ({_z} + 74), {_w}) and location({_bx2}, 280, ({_z} + 75), {_w}) to smooth_stone

    if {_plotX} < {_lastIndex}:

function mauPlotBedrockDepth(plotKey: text) :: number:
    set {_depth} to {mauplot::plot-bedrock-depth::%{_plotKey}%}
    if {_depth} is not set:
        set {_depth} to {mauplot::bedrock-default-depth}
    if {_depth} is not set:
        set {_depth} to 0
    return {_depth}

function mauPlotQueueBedrockRegion(id: text, w: world, x1: number, z1: number, x2: number, z2: number, depth: number):
    if {_depth} <= 0:
        stop

    # Bereits fertige Bereiche werden nur dann neu aufgebaut, wenn sich die Tiefe geändert hat.
    if {mauplot::bedrock-ready-depth::%{_id}%} is {_depth}:
        stop

    # Wenn die Region schon in der Warteschlange steht, nur bei geänderter Tiefe neu starten.
    if {mauplot::bedrock-queued::%{_id}%} is true:
        if {mauplot::bedrock-data::depth::%{_id}%} is {_depth}:
            stop
        delete {mauplot::bedrock-ready::%{_id}%}
        set {mauplot::bedrock-data::depth::%{_id}%} to {_depth}
        set {mauplot::bedrock-y::%{_id}%} to {@mau-world-surface} - {_depth}
        set {mauplot::bedrock-data::phase::%{_id}%} to "floor"
        set {mauplot::bedrock-data::tile::%{_id}%} to 0
        set {mauplot::bedrock-data::layer::%{_id}%} to 0
        stop

    if {mauplot::bedrock-queue-head} is not set:
        set {mauplot::bedrock-queue-head} to 1
    if {mauplot::bedrock-queue-tail} is not set:
        set {mauplot::bedrock-queue-tail} to 0

    add 1 to {mauplot::bedrock-queue-tail}
    set {_queueIndex} to {mauplot::bedrock-queue-tail}
    set {mauplot::bedrock-queue::%{_queueIndex}%} to {_id}
    set {mauplot::bedrock-queue-index::%{_id}%} to {_queueIndex}
    set {mauplot::bedrock-queued::%{_id}%} to true
    set {mauplot::bedrock-data::world::%{_id}%} to {_w}
    set {mauplot::bedrock-data::x1::%{_id}%} to {_x1}
    set {mauplot::bedrock-data::z1::%{_id}%} to {_z1}
    set {mauplot::bedrock-data::x2::%{_id}%} to {_x2}
    set {mauplot::bedrock-data::z2::%{_id}%} to {_z2}
    set {mauplot::bedrock-data::depth::%{_id}%} to {_depth}
    set {mauplot::bedrock-y::%{_id}%} to {@mau-world-surface} - {_depth}
    delete {mauplot::bedrock-ready::%{_id}%}

    set {_width} to {_x2} - {_x1} + 1
    set {_regionDepth} to {_z2} - {_z1} + 1
    set {mauplot::bedrock-data::tiles-x::%{_id}%} to floor(({_width} + 15) / 16)
    set {mauplot::bedrock-data::tiles-z::%{_id}%} to floor(({_regionDepth} + 15) / 16)
    set {mauplot::bedrock-data::phase::%{_id}%} to "floor"
    set {mauplot::bedrock-data::tile::%{_id}%} to 0
    set {mauplot::bedrock-data::layer::%{_id}%} to 0

    if {mauplot::bedrock-queue::active} is not set:
        set {mauplot::bedrock-queue::active} to {_id}

function mauPlotFinishBedrockRegion(id: text):
    set {_completedDepth} to {mauplot::bedrock-data::depth::%{_id}%}
    set {mauplot::bedrock-ready::%{_id}%} to true
    set {mauplot::bedrock-ready-depth::%{_id}%} to {_completedDepth}
    delete {mauplot::bedrock-queued::%{_id}%}
    delete {mauplot::bedrock-data::world::%{_id}%}
    delete {mauplot::bedrock-data::x1::%{_id}%}
    delete {mauplot::bedrock-data::z1::%{_id}%}
    delete {mauplot::bedrock-data::x2::%{_id}%}
    delete {mauplot::bedrock-data::z2::%{_id}%}
    delete {mauplot::bedrock-data::tiles-x::%{_id}%}
    delete {mauplot::bedrock-data::tiles-z::%{_id}%}
    delete {mauplot::bedrock-data::depth::%{_id}%}
    delete {mauplot::bedrock-data::phase::%{_id}%}
    delete {mauplot::bedrock-data::tile::%{_id}%}
    delete {mauplot::bedrock-data::layer::%{_id}%}

    if {mauplot::bedrock-queue-head} is set:
        delete {mauplot::bedrock-queue::%{mauplot::bedrock-queue-head}%}
        add 1 to {mauplot::bedrock-queue-head}

    delete {mauplot::bedrock-queue::active}
    if {mauplot::bedrock-queue::%{mauplot::bedrock-queue-head}%} is set:
        set {mauplot::bedrock-queue::active} to {mauplot::bedrock-queue::%{mauplot::bedrock-queue-head}%}
    else:
        set {mauplot::bedrock-queue-head} to 1
        set {mauplot::bedrock-queue-tail} to 0


function mauPlotMergeRoad(plotX: number, plotZ: number, axis: text, w: world):
    # plotX/plotZ bezeichnen immer das westliche bzw. nördliche Plot der Verbindung.
    set {_baseX} to {mauplot::originX} + ({_plotX} * {mauplot::grid-step})
    set {_baseZ} to {mauplot::originZ} + ({_plotZ} * {mauplot::grid-step})
    if {_axis} is "x":
        set {_roadX1} to {_baseX} + 64
        set {_roadX2} to {_baseX} + 75
        set {_roadZ1} to {_baseZ}
        set {_roadZ2} to {_baseZ} + 63
    if {_axis} is "z":
        set {_roadX1} to {_baseX}
        set {_roadX2} to {_baseX} + 63
        set {_roadZ1} to {_baseZ} + 64
        set {_roadZ2} to {_baseZ} + 75
    # Straßenstreifen sicher und ohne WorldEdit in bebaubares Gelände umwandeln.
    set all blocks within location({_roadX1}, 279, {_roadZ1}, {_w}) and location({_roadX2}, 279, {_roadZ2}, {_w}) to bedrock
    set all blocks within location({_roadX1}, 280, {_roadZ1}, {_w}) and location({_roadX2}, 280, {_roadZ2}, {_w}) to grass_block
    set all blocks within location({_roadX1}, 281, {_roadZ1}, {_w}) and location({_roadX2}, 320, {_roadZ2}, {_w}) to air

    # Eine geöffnete Merge-Straße erhält die Tiefe des angrenzenden Plot-Standards.
    set {_mergePlotKey} to "%{_w}%::%{_plotX}%::%{_plotZ}%"
    set {_mergeDepth} to mauPlotBedrockDepth({_mergePlotKey})
    if {_mergeDepth} > 0:
        set {_mergeId} to "%{_w}%::MERGE::%{_plotX}%::%{_plotZ}%::%{_axis}%"
        mauPlotQueueBedrockRegion({_mergeId}, {_w}, {_roadX1}, {_roadZ1}, {_roadX2}, {_roadZ2}, {_mergeDepth})

function mauPlotSendInfo(p: player, k: text):
    send "&8&m--------------------------------" to {_p}
    send "&b&lINFOS" to {_p}

    if {mauplot::owner::%{_k}%} is not set:
        send "&7Besitzer: &fNiemand" to {_p}
        send "&7Vertraute: &f0" to {_p}
        send "&7Mitglieder: &f0" to {_p}
        send "&7Erstellt am: &fNoch nicht beansprucht" to {_p}
        send "&8&m--------------------------------" to {_p}
        send "&7Nutze &e/p a &7zum Beanspruchen." to {_p}
        stop

    set {_owner} to {mauplot::ownername::%{_k}%}
    if {_owner} is not set:
        set {_owner} to "Unbekannt"

    set {_trustedCount} to size of {mauplot::trusted::%{_k}%::*}
    set {_addedCount} to size of {mauplot::added::%{_k}%::*}

    set {_trustedNames} to ""
    loop {mauplot::trusted-name::%{_k}%::*}:
        if {_trustedNames} is "":
            set {_trustedNames} to loop-value
        else:
            set {_trustedNames} to "%{_trustedNames}%, %loop-value%"
    if {_trustedNames} is "":
        if {_trustedCount} > 0:
            set {_trustedNames} to "Namen nicht gespeichert"
        else:
            set {_trustedNames} to "Keine"

    set {_addedNames} to ""
    loop {mauplot::added-name::%{_k}%::*}:
        if {_addedNames} is "":
            set {_addedNames} to loop-value
        else:
            set {_addedNames} to "%{_addedNames}%, %loop-value%"
    if {_addedNames} is "":
        if {_addedCount} > 0:
            set {_addedNames} to "Namen nicht gespeichert"
        else:
            set {_addedNames} to "Keine"

    send "&7Besitzer: &e%{_owner}%" to {_p}
    send "&7Vertraute: &f%{_trustedNames}% &8(%{_trustedCount}%)" to {_p}
    send "&7Mitglieder: &f%{_addedNames}% &8(%{_addedCount}%)" to {_p}

    if {mauplot::created::%{_k}%} is set:
        set {_created} to {mauplot::created::%{_k}%}
        send "&7Erstellt am: &f%{_created}%" to {_p}
    else:
        send "&7Erstellt am: &fUnbekannt (vor dem Update erstellt)" to {_p}

    send "&8&m--------------------------------" to {_p}

                stop

                send "&cDu kannst dir dein eigenes Plot nicht vergeben." to player
                stop

            set {_targetMaximum} to mauPlotMax({_target})

                send "&eDieser Spieler besitzt das Plot bereits." to player
                stop
            if {_targetCount} >= {_targetMaximum}:
                send "&c%name of {_target}% hat bereits das Maximum von %{_targetMaximum}% Plots." to player
                stop
            if {mauplot::locked::%{_key}%} is set:
                send "&cDieses Plot ist gesperrt und kann nicht vergeben werden." to player
                stop

                stop

                stop

            set {_parts::*} to {_key} split at "::"
            set {_plotX} to {_parts::2} parsed as number
            set {_plotZ} to {_parts::3} parsed as number
            set {_direction} to arg-2

            if {_direction} is not set:

                set {_pairZ} to {_plotZ}

                set {_pairX} to {_otherX}
                set {_pairZ} to {_plotZ}

                set {_validDirection} to true
                set {_otherX} to {_plotX}
                set {_otherZ} to {_plotZ} + 1
                set {_pairX} to {_plotX}
                set {_pairZ} to {_plotZ}
                set {_axis} to "z"

                set {_otherX} to {_plotX}
                set {_otherZ} to {_plotZ} - 1

                set {_pairZ} to {_otherZ}
                set {_axis} to "z"
            if {_validDirection} is false:
                send "&cRichtung: north, south, east oder west. Beispiel: /p merge east" to player

                send "&cAußerhalb des Plotrasters." to player
                stop
            if {_otherZ} < 0:
                send "&cAußerhalb des Plotrasters." to player
                stop
            if {_otherX} >= {mauplot::grid-count}:
                send "&cAußerhalb des Plotrasters." to player
                stop
            if {_otherZ} >= {mauplot::grid-count}:
                send "&cAußerhalb des Plotrasters." to player
                stop

            set {_neighborKey} to "%{mauplot::world}%::%{_otherX}%::%{_otherZ}%"
            if {mauplot::owner::%{_neighborKey}%} is not {_uuid}:
                send "&cDas angrenzende Plot gehört dir nicht. Beide Plots müssen dir gehören." to player
                stop
            if {mauplot::locked::%{_neighborKey}%} is set:
                send "&cDas Nachbar-Plot ist gesperrt." to player
                stop

            if {_axis} is "x":
                set {_pairBX} to {_pairX} + 1
                set {_pairB} to "%{mauplot::world}%::%{_pairBX}%::%{_pairZ}%"
            if {_axis} is "z":

                set {_pairB} to "%{mauplot::world}%::%{_pairX}%::%{_pairBZ}%"
            if {mauplot::merge::%{_pairA}%::%{_pairB}%} is true:
                send "&eDiese beiden Grundstücke sind bereits gemergt." to player
                stop

            mauPlotGenerateCell({_pairX}, {_pairZ}, {mauplot::world})
            mauPlotGenerateCell({_otherX}, {_otherZ}, {mauplot::world})
            set {mauplot::merge::%{_pairA}%::%{_pairB}%} to true
            set {mauplot::merge::%{_pairB}%::%{_pairA}%} to true
            add {_pairB} to {mauplot::merged-with::%{_pairA}%::*}
            add {_pairA} to {mauplot::merged-with::%{_pairB}%::*}
            mauPlotMergeRoad({_pairX}, {_pairZ}, {_axis}, {mauplot::world})
            send "&aErfolgreich gemergt! &7Richtung: &e%{_direction}%&7. Die Straße zwischen den Plots wurde entfernt." to player
            stop

        if {_sub} is "a" or "claim":
            set {_key} to mauPlotKey(location of player)
            if {_key} is "OUTSIDE":
                send "&cDiese Welt ist nicht als Plotwelt eingerichtet. Admin: /p setworld" to player
                stop
            if {mauplot::owner::%{_key}%} is set:
                if {mauplot::owner::%{_key}%} is {_uuid}:
                    send "&eDieses Plot gehört bereits dir. Nutze &f/p h&e." to player

                    send "&cDieses Plot gehört bereits jemand anderem." to player
                stop
            if {mauplot::locked::%{_key}%} is set:

                stop

            set {_maximum} to mauPlotMax(player)
            if {_count} >= {_maximum}:
                send "&cDu hast bereits dein Maximum von &e%{_maximum}% &cPlots." to player
                stop

            set {_claimZ} to {_plotParts::3} parsed as number

            add {_key} to {mauplot::playerplots::%{_uuid}%::*}
            if {mauplot::playerplot::%{_uuid}%} is not set:
                set {mauplot::playerplot::%{_uuid}%} to {_key}
            set {mauplot::home::%{_key}%} to mauPlotHomeLocation({_claimX}, {_claimZ}, {mauplot::world})
            send "&aDu hast dieses &e64x64 &aPlot beansprucht!" to player
            send "&7Nutze &e/p auto &7für ein automatisch gefundenes Plot." to player

                stop

            if {mauplot::grid-enabled} is true:
                set {_originX} to floor(((x-coordinate of player) - {mauplot::originX}) / {mauplot::grid-step})
                set {_originZ} to floor(((z-coordinate of player) - {mauplot::originZ}) / {mauplot::grid-step})
            else:
                set {_originX} to floor((x-coordinate of player) / {@plot-size})
                set {_originZ} to floor((z-coordinate of player) / {@plot-size})
            set {_found} to false
            set {_radius} to 0

            # Ringweise Suche: zuerst das aktuelle Feld, dann die direkten Nachbarn usw.
            # Dadurch werden nur die Randfelder jedes Rings geprüft.
            loop numbers from 0 to {@plot-auto-radius}:
                if {_found} is false:
                    set {_radius} to loop-number
                    set {_minOffset} to 0 - {_radius}
                    set {_maxOffset} to {_radius}

                    # Untere und obere Kante des Rings.
                    loop numbers from {_minOffset} to {_maxOffset}:
                        if {_found} is false:
                            set {_xOffset} to loop-value-1
                            set {_candidateX} to {_originX} + {_xOffset}
                            set {_candidateZ} to {_originZ} + {_minOffset}
                            if mauPlotCellAvailable({_candidateX}, {_candidateZ}, {mauplot::world}) is true:
                                set {_found} to true
                                set {_foundX} to {_candidateX}
                                set {_foundZ} to {_candidateZ}
                                set {_foundKey} to "%{mauplot::world}%::%{_foundX}%::%{_foundZ}%"
                            if {_found} is false:
                                if {_radius} > 0:
                                    set {_candidateZ} to {_originZ} + {_maxOffset}
                                    if mauPlotCellAvailable({_candidateX}, {_candidateZ}, {mauplot::world}) is true:
                                        set {_found} to true
                                        set {_foundX} to {_candidateX}
                                        set {_foundZ} to {_candidateZ}
                                        set {_foundKey} to "%{mauplot::world}%::%{_foundX}%::%{_foundZ}%"

                    # Linke und rechte Kante ohne Ecken, damit keine Doppelprüfungen entstehen.
                    if {_found} is false:
                        if {_radius} > 0:
                            set {_innerMin} to {_minOffset} + 1
                            set {_innerMax} to {_maxOffset} - 1
                            loop numbers from {_innerMin} to {_innerMax}:
                                if {_found} is false:
                                    set {_zOffset} to loop-value-1
                                    set {_candidateZ} to {_originZ} + {_zOffset}
                                    set {_candidateX} to {_originX} + {_minOffset}
                                    if mauPlotCellAvailable({_candidateX}, {_candidateZ}, {mauplot::world}) is true:
                                        set {_found} to true
                                        set {_foundX} to {_candidateX}
                                        set {_foundZ} to {_candidateZ}
                                        set {_foundKey} to "%{mauplot::world}%::%{_foundX}%::%{_foundZ}%"
                                    if {_found} is false:
                                        set {_candidateX} to {_originX} + {_maxOffset}
                                        if mauPlotCellAvailable({_candidateX}, {_candidateZ}, {mauplot::world}) is true:
                                            set {_found} to true
                                            set {_foundX} to {_candidateX}
                                            set {_foundZ} to {_candidateZ}
                                            set {_foundKey} to "%{mauplot::world}%::%{_foundX}%::%{_foundZ}%"

                send "&cIn deiner Nähe wurde kein freies Plot gefunden. Versuche es an einer anderen Stelle." to player

            stop

                close player's inventory
                make player execute command "/p auto"
            else if clicked slot is 14:
                close player's inventory
                make player execute command "/p h"
            else if clicked slot is 16:
                close player's inventory
                make player execute command "/p list"

                close player's inventory

on player move:
    if {mauplot::world} is not set:
        stop
    if world of future event-location is not {mauplot::world}:
        delete {mauplot::deny-notice::%uuid of player%}
        stop

    set {_fromKey} to mauPlotKey(past event-location)
    set {_toKey} to mauPlotKey(future event-location)
    if {_toKey} is {_fromKey}:
        stop

    if {_toKey} is not "OUTSIDE":
        if {mauplot::denied::%{_toKey}%::%uuid of player%} is set:
            if mauPlotIsAdmin(player) is false:
                cancel event

command /shop [<text>] [<number>] [<number>]:
    executable by: players
    trigger:
        set {_uuid} to "%uuid of player%"

        if arg-1 is not set:
            send "&8&m------------------------------" to player
            send "&d&lMAU Spielershops" to player
            send "&e/shop create <preis> [menge] &7- Shop mit Item in der Hand erstellen" to player
            send "&e/shop remove &7- Eigenen Shop entfernen" to player
            send "&e/shop info &7- Shop-Informationen anzeigen" to player
            send "&7Sneak-Rechtsklick auf die eigene Shop-Kiste öffnet den Vorrat." to player
            send "&7Normaler Rechtsklick kauft die eingestellte Menge." to player
            send "&8&m------------------------------" to player
            stop

        if arg-1 is "help":
            send "&8&m------------------------------" to player
            send "&d&lMAU Spielershops" to player
            send "&e/shop create <preis> [menge] &7- Shop mit Item in der Hand erstellen" to player
            send "&e/shop remove &7- Eigenen Shop entfernen" to player
            send "&e/shop info &7- Shop-Informationen anzeigen" to player
            send "&7Sneak-Rechtsklick auf die eigene Shop-Kiste öffnet den Vorrat." to player
            send "&7Normaler Rechtsklick kauft die eingestellte Menge." to player
            send "&8&m------------------------------" to player
            stop

        if arg-1 is "create":
            if arg-2 is not set:
                send "{@shop-prefix}&cNutze: /shop create <preis> [menge]" to player
                stop
            if arg-2 <= 0:
                send "{@shop-prefix}&cDer Preis muss größer als 0 sein." to player
                stop
            if arg-2 > {@shop-max-price}:
                send "{@shop-prefix}&cDer Preis ist zu hoch." to player
                stop
            if player's tool is air:
                send "{@shop-prefix}&cHalte das Item in der Hand, das du verkaufen möchtest." to player
                stop

            set {_amount} to 1
            if arg-3 is set:
                set {_amount} to arg-3
            if {_amount} < 1:
                send "{@shop-prefix}&cDie Menge muss mindestens 1 sein." to player
                stop
            if {_amount} > {@shop-max-amount}:
                send "{@shop-prefix}&cMaximal {@shop-max-amount} Items pro Kauf." to player
                stop

            set {_template} to player's tool
            set amount of {_template} to 1
            set {maushop::setup::price::%{_uuid}%} to arg-2
            set {maushop::setup::amount::%{_uuid}%} to {_amount}
            set {maushop::setup::item::%{_uuid}%} to {_template}
            send "{@shop-prefix}&aRechtsklicke jetzt auf eine freie Kiste, um deinen Shop zu erstellen." to player
            stop

        if arg-1 is "remove":
            set {maushop::setup::remove::%{_uuid}%} to true
            send "{@shop-prefix}&eRechtsklicke auf deine Shop-Kiste, um sie zu entfernen." to player
            stop

        if arg-1 is "info":
            set {maushop::setup::info::%{_uuid}%} to true
            send "{@shop-prefix}&eRechtsklicke auf eine Shop-Kiste, um ihre Daten anzuzeigen." to player
            stop

        send "{@shop-prefix}&cUnbekannter Unterbefehl. Nutze /shop help." to player

on rightclick:
    # Nur Klicks auf tatsächlich vorhandene Blöcke weiterverarbeiten.
    if event-block is not set:
        stop
    if event-block is not chest:
        if event-block is not trapped chest:
            stop

    set {_key} to "%location of event-block%"
    set {_uuid} to "%uuid of player%"

    # Shop erstellen
    if {maushop::setup::price::%{_uuid}%} is set:
        cancel event
        if {maushop::owner::%{_key}%} is set:
            send "{@shop-prefix}&cDiese Kiste ist bereits ein Shop." to player
            stop
        set {maushop::owner::%{_key}%} to {_uuid}
        set {maushop::ownername::%{_key}%} to name of player
        set {maushop::item::%{_key}%} to {maushop::setup::item::%{_uuid}%}
        set {maushop::price::%{_key}%} to {maushop::setup::price::%{_uuid}%}
        set {maushop::amount::%{_key}%} to {maushop::setup::amount::%{_uuid}%}
        delete {maushop::setup::price::%{_uuid}%}
        delete {maushop::setup::amount::%{_uuid}%}
        delete {maushop::setup::item::%{_uuid}%}
        send "{@shop-prefix}&aShop erstellt! Lege den Verkaufsbestand in die Kiste." to player
        send "{@shop-prefix}&7Preis: &e%{maushop::price::%{_key}%}%$ &8| &7Menge: &e%{maushop::amount::%{_key}%}%" to player
        stop

    # Shop entfernen
    if {maushop::setup::remove::%{_uuid}%} is set:
        cancel event
        if {maushop::owner::%{_key}%} is not set:
            send "{@shop-prefix}&cDas ist keine Shop-Kiste." to player
            stop
        if {maushop::owner::%{_key}%} is not {_uuid}:
            send "{@shop-prefix}&cDas ist nicht dein Shop." to player
            stop
        delete {maushop::owner::%{_key}%}
        delete {maushop::ownername::%{_key}%}
        delete {maushop::item::%{_key}%}
        delete {maushop::price::%{_key}%}
        delete {maushop::amount::%{_key}%}
        delete {maushop::setup::remove::%{_uuid}%}
        send "{@shop-prefix}&aShop entfernt. Die Kiste ist wieder normal." to player
        stop

    # Shop-Informationen
    if {maushop::setup::info::%{_uuid}%} is set:
        cancel event
        if {maushop::owner::%{_key}%} is not set:
            send "{@shop-prefix}&cDas ist keine Shop-Kiste." to player
            stop
        send "&8&m------------------------------" to player
        send "&d&lShop-Information" to player
        send "&7Besitzer: &e%{maushop::ownername::%{_key}%}%" to player
        send "&7Item: &e%{maushop::item::%{_key}%}%" to player
        send "&7Preis: &a%{maushop::price::%{_key}%}%$" to player
        send "&7Menge: &e%{maushop::amount::%{_key}%}%" to player
        send "&8&m------------------------------" to player
        delete {maushop::setup::info::%{_uuid}%}
        stop

    # Normale Kiste: nichts verändern
    if {maushop::owner::%{_key}%} is not set:
        stop

    set {_sellerID} to {maushop::owner::%{_key}%}

    # Shop-Besitzer kann mit Shift-Rechtsklick den Vorrat öffnen.
    if player is sneaking:
        if {_sellerID} is {_uuid}:
            stop

    cancel event

    if {_sellerID} is {_uuid}:
        send "{@shop-prefix}&7Halte Shift und rechtsklicke, um den Vorrat zu öffnen." to player
        stop

    if {maudata::db} is not set:
        send "{@shop-prefix}&cDie gemeinsame Datenbank ist gerade nicht verfügbar." to player
        stop

    set {_template} to {maushop::item::%{_key}%}
    set {_quantity} to {maushop::amount::%{_key}%}
    set {_price} to {maushop::price::%{_key}%}
    set {_totalPrice} to {_price}

    if {_template} is not set:
        send "{@shop-prefix}&cDieser Shop ist fehlerhaft konfiguriert. Bitte melde dich beim Team." to player
        stop
    if {_quantity} is not set:
        send "{@shop-prefix}&cDieser Shop ist fehlerhaft konfiguriert. Bitte melde dich beim Team." to player
        stop
    if {_price} is not set:
        send "{@shop-prefix}&cDieser Shop ist fehlerhaft konfiguriert. Bitte melde dich beim Team." to player
        stop

    # Bestand im Shopinventar prüfen (nach Materialtyp).
    set {_shopInventory} to inventory of event-block
    set {_slotCount} to amount of slots of {_shopInventory}
    set {_stock} to 0
    loop {_slotCount} times:
        set {_slotNumber} to loop-number - 1
        set {_stack} to slot {_slotNumber} of {_shopInventory}
        if {_stack} is not air:
            if type of {_stack} is type of {_template}:
                add item amount of {_stack} to {_stock}

    if {_stock} < {_quantity}:
        send "{@shop-prefix}&cDieser Shop ist momentan ausverkauft." to player
        stop

    set {_buyerID} to {_uuid}
    if {_buyerID} is {_sellerID}:
        send "{@shop-prefix}&cDu kannst nicht bei deinem eigenen Shop kaufen." to player
        stop

    # Konten sicherstellen und Käuferguthaben abfragen.
    execute "INSERT IGNORE INTO mau_money (player_uuid, balance) VALUES (?, 0)" in {maudata::db} with arguments {_buyerID}
    execute "INSERT IGNORE INTO mau_money (player_uuid, balance) VALUES (?, 0)" in {maudata::db} with arguments {_sellerID}

    delete {_balanceData::*}
    execute "SELECT balance FROM mau_money WHERE player_uuid = ?" in {maudata::db} with arguments {_buyerID} and store the result in {_balanceData::*}

    if {_balanceData::balance::1} is not set:
        send "{@shop-prefix}&cDein Kontostand konnte nicht geladen werden." to player
        stop

    set {_balance} to {_balanceData::balance::1}
    if {_balance} < {_totalPrice}:
        send "{@shop-prefix}&cDu hast nicht genug Geld. Preis: &e%{_totalPrice}%$" to player
        stop

    # Geldtransfer atomar in einer SQL-Anweisung.
    delete {_changed}
    execute "UPDATE mau_money AS payer JOIN mau_money AS receiver ON receiver.player_uuid = ? SET payer.balance = payer.balance - ?, receiver.balance = receiver.balance + ? WHERE payer.player_uuid = ? AND payer.balance >= ?" in {maudata::db} with arguments {_sellerID} and {_totalPrice} and {_totalPrice} and {_buyerID} and {_totalPrice} and store the result in {_changed}

    if last sql error is set:
        send "{@shop-prefix}&cDie Zahlung ist fehlgeschlagen. Bitte versuche es erneut." to player
        send "[MAU-Shop] SQL-Fehler: %last sql error%" to console
        stop

    if {_changed} is not set:
        send "{@shop-prefix}&cDie Zahlung konnte nicht bestätigt werden. Bitte versuche es erneut." to player
        stop
    if {_changed} < 2:
        send "{@shop-prefix}&cDein Kontostand hat sich geändert. Bitte versuche es erneut." to player
        stop

    # Bestand nach bestätigter Zahlung entnehmen und ausgeben.
    remove {_quantity} of type of {_template} from {_shopInventory}
    set {_deliver} to {_template}
    set amount of {_deliver} to {_quantity}
    give {_deliver} to player

    # JobGeld-Cache des Käufers und eines online befindlichen Verkäufers aktualisieren.
    set {jobgeld::cache::money::%{_buyerID}%} to {_balance} - {_totalPrice}
    loop all players:
        if "%uuid of loop-player%" is {_sellerID}:
            if {jobgeld::cache::money::%{_sellerID}%} is set:
                add {_totalPrice} to {jobgeld::cache::money::%{_sellerID}%}
            send "{@shop-prefix}&aDein Shop hat einen Verkauf für &e%{_totalPrice}%$ &agemacht." to loop-player

    send "{@shop-prefix}&aKauf erfolgreich! Du hast &e%{_quantity}%x %{_template}% &afür &e%{_totalPrice}%$ &agekauft." to player

function mauCbSpawnProtected(l: location) :: boolean:
    if {maucb::spawn} is not set:
        return false
    if world of {_l} is not world of {maucb::spawn}:
        return false
    if distance between {_l} and {maucb::spawn} <= {@spawn-protection-radius}:
        return true
    return false

    executable by: players
    trigger:
        set {maucb::spawn} to location of player
        send "{@cb-prefix}&aDer CityBuild-Spawn wurde gesetzt." to player

command /cbspawn:
    executable by: players
    trigger:
        if {maucb::spawn} is not set:
            send "{@cb-prefix}&cDer Spawn wurde noch nicht eingerichtet." to player
            stop
        teleport player to {maucb::spawn}
        send "{@cb-prefix}&aDu wurdest zum CityBuild-Spawn teleportiert." to player

command /cbsetwarp <text>:
    permission: maucb.admin
    executable by: players
    trigger:
        set {_name} to arg-1
        if {_name} contains " ":
            send "{@cb-prefix}&cWarp-Namen dürfen keine Leerzeichen enthalten." to player
            stop
        set {maucb::warp::%{_name}%} to location of player
        if {maucb::warp-list::*} does not contain {_name}:
            add {_name} to {maucb::warp-list::*}
        send "{@cb-prefix}&aWarp &e%{_name}% &awurde gesetzt." to player

command /cbdelwarp <text>:
    permission: maucb.admin
    executable by: players
    trigger:
        set {_name} to arg-1
        if {maucb::warp::%{_name}%} is not set:
            send "{@cb-prefix}&cDieser Warp existiert nicht." to player
            stop
        delete {maucb::warp::%{_name}%}
        remove {_name} from {maucb::warp-list::*}
        send "{@cb-prefix}&aWarp &e%{_name}% &awurde gelöscht." to player

# Kein Multiverse-Core, kein FAWE, keine WorldEdit-Befehle.
# /cb welt erstellen  - erstellt die Welt oder setzt eine unterbrochene
#                       Generierung ab dem gespeicherten Fortschritt fort.
# /cb welt status     - zeigt den Stand.
#
# Design: 100x100 Plot-Raster (10.000 Grundstücke), 64x64 je Plot,
# 12 Blöcke breite Straßen. Die zentrale 6x6-Startstadt wird vorgebaut;
# weitere Plotflächen und Straßenstücke werden beim ersten Claim angelegt.
#
# Die Welt mau_citybuild_01 wird als eigene normale Welt geladen.
# 10.000 Plotzellen liegen in einem 100x100-Raster. Die Startstadt bleibt
# mittig erhalten; weitere Zellen werden erst bei der ersten Beanspruchung
# gebaut. Worldborder: 7.600 x 7.600 Blöcke (Größe/Durchmesser 7.600).
# Kern-Skript kann keinen eigenen Flat-World-Generator konfigurieren;
# die Grundstücke liegen wie die Startstadt auf Y=280.

function mauCbGenTree(w: world, x: number, z: number):
    set {_x1} to {_x} - 2
    set {_x2} to {_x} + 2
    set {_z1} to {_z} - 2
    set {_z2} to {_z} + 2
    set {_tx1} to {_x} - 1
    set {_tx2} to {_x} + 1
    set {_tz1} to {_z} - 1
    set {_tz2} to {_z} + 1

    # Kleiner Pflanzbereich auf dem Gehweg.
    set all blocks within location({_x1}, 280, {_z1}, {_w}) and location({_x2}, 280, {_z2}, {_w}) to grass_block
    # Baumkrone in zwei breiten Etagen plus kleiner Spitze.
    set all blocks within location({_x1}, 284, {_z1}, {_w}) and location({_x2}, 285, {_z2}, {_w}) to oak_leaves
    set all blocks within location({_tx1}, 286, {_tz1}, {_w}) and location({_tx2}, 286, {_tz2}, {_w}) to oak_leaves
    # Stamm nach den Blättern setzen, damit er sichtbar bleibt.
    set block at location({_x}, 281, {_z}, {_w}) to oak_log
    set block at location({_x}, 282, {_z}, {_w}) to oak_log
    set block at location({_x}, 283, {_z}, {_w}) to oak_log
    set block at location({_x}, 284, {_z}, {_w}) to oak_log
    set block at location({_x}, 285, {_z}, {_w}) to oak_log

function mauCbGenLamp(w: world, x: number, z: number):
    set block at location({_x}, 280, {_z}, {_w}) to stone_bricks
    set block at location({_x}, 281, {_z}, {_w}) to dark_oak_fence
    set block at location({_x}, 282, {_z}, {_w}) to dark_oak_fence
    set block at location({_x}, 283, {_z}, {_w}) to dark_oak_fence
    set block at location({_x}, 284, {_z}, {_w}) to dark_oak_fence
    set block at location({_x}, 285, {_z}, {_w}) to sea_lantern
    set block at location({_x}, 286, {_z}, {_w}) to sea_lantern

command /cb [<text>] [<text>] [<text>]:
    executable by: players
    trigger:
        # OPs und ausdrücklich berechtigte CityBuild-Admins dürfen den Generator nutzen.
        if mauPlotIsAdmin(player) is false:
            send "&cDafür hast du keine Berechtigung. Dieser Befehl ist nur für OPs und CityBuild-Admins." to player
            stop

        if argument 1 is "welt":
            if argument 2 is "status":
                if {mauworldgen::state} is not set:
                    send "&b[MAU-Generator] &7Noch nicht gestartet. Nutze &e/cb welt erstellen&7." to player
                else if {mauworldgen::state} is "done":
                    send "&b[MAU-Generator] &aDie CityBuild-Welt ist fertig." to player
                    send "&7Kapazität: &e10.000 Plots (100x100) &8| &7Worldborder: &e7.600 x 7.600" to player
                else:
                    send "&b[MAU-Generator] &eStatus: %{mauworldgen::state}% &8| &ePhase: %{mauworldgen::stage}% &8| &eFortschritt: %{mauworldgen::progress}%" to player
                stop

        if argument 1 is not "welt":
            send "&8&m--------------------------------" to player
            send "&b&lMAU CITYBUILD GENERATOR" to player
            send "&e/cb welt erstellen &7- Neue Welt erstellen / gespeicherten Lauf fortsetzen" to player
            send "&e/cb welt erstellen bestaetigen &7- Vorhandene Welt nach Warnung aktivieren/fortsetzen" to player
            send "&e/cb welt status &7- Fortschritt anzeigen" to player
            send "&8&m--------------------------------" to player
            stop

        if argument 2 is not set:
            send "&8&m--------------------------------" to player
            send "&b&lMAU CITYBUILD GENERATOR" to player
            send "&e/cb welt erstellen &7- Welt erstellen / Generierung fortsetzen" to player
            send "&e/cb welt status &7- Fortschritt anzeigen" to player
            send "&8&m--------------------------------" to player
            stop

        if argument 2 is not "erstellen":
            send "&8&m--------------------------------" to player
            send "&b&lMAU CITYBUILD GENERATOR" to player
            send "&e/cb welt erstellen &7- Welt erstellen / Generierung fortsetzen" to player
            send "&e/cb welt status &7- Fortschritt anzeigen" to player
            send "&8&m--------------------------------" to player
            stop

        if {mauplot::recovery-required} is true:
            send "&c[MAU-Generator] Die gespeicherte CityBuild-Welt wurde nicht sicher erkannt." to player
            send "&cBitte Weltordner/Backup prüfen. Eine neue Generierung wird aus Sicherheitsgründen blockiert." to player
            stop

        if {mauworldgen::active} is true:
            send "&e[MAU-Generator] Eine Generierung läuft bereits. Nutze /cb welt status." to player
            stop

        # Dritte Angabe muss ausdrücklich bestaetigen/confirm lauten.
        set {_confirmed} to false
        if argument 3 is "bestaetigen":
            set {_confirmed} to true
        if argument 3 is "confirm":
            set {_confirmed} to true

        if {_confirmed} is true:
            if {mauworldgen::confirm::%uuid of player%} is not true:
                send "&cDu hast keine offene Bestätigungsanfrage. Gib zuerst /cb welt erstellen ein." to player
                stop
            delete {mauworldgen::confirm::%uuid of player%}

            # Eine fertige Welt wird niemals neu überschrieben. Bestätigung bedeutet:
            # bestehende Welt laden, Layout prüfen und an Plot/Spawn/Border anbinden.
            if {mauworldgen::state} is "done":
                set {_existingWorld} to world named "{@mau-world-name}"
                if {_existingWorld} is not set:
                    load world "{@mau-world-name}" with environment normal
                    wait 3 seconds
                    set {_existingWorld} to world named "{@mau-world-name}"
                if {_existingWorld} is not set:
                    set {mauplot::recovery-required} to true
                    send "&c[MAU-Generator] Die vorhandene Weltdatei konnte nicht geladen werden. Es wurde nichts neu erstellt." to player
                    stop

                if block at location(-240, 278, -240, {_existingWorld}) is not bedrock:
                    set {mauplot::recovery-required} to true
                    send "&c[MAU-Generator] Der Layout-Fingerabdruck stimmt nicht. Es wurde nichts verändert." to player
                    stop
                if block at location(-240, 279, -240, {_existingWorld}) is not stone:
                    set {mauplot::recovery-required} to true
                    send "&c[MAU-Generator] Der Layout-Fingerabdruck stimmt nicht. Es wurde nichts verändert." to player
                    stop
                if block at location(-240, 280, -240, {_existingWorld}) is not smooth stone:
                    set {mauplot::recovery-required} to true
                    send "&c[MAU-Generator] Der Layout-Fingerabdruck stimmt nicht. Es wurde nichts verändert." to player
                    stop

                set {mauplot::world} to {_existingWorld}
                set {mauplot::grid-enabled} to true
                set {mauplot::originX} to -3794
                set {mauplot::originZ} to -3794
                set {mauplot::grid-step} to 76
                set {mauplot::grid-count} to 100
                set {maucb::citybuild-world} to {_existingWorld}
                set {_border} to world border of {_existingWorld}
                set world border radius of {_border} to 7600
                set world border center of {_border} to location(6, 280, 6, {_existingWorld})
                set world border warning distance of {_border} to 25
                set world border damage buffer of {_border} to 5
                set worldborder of {_existingWorld} to {_border}
                if {maucb::spawn} is not set:
                    set {maucb::spawn} to location(6, 281, 6, {_existingWorld})
                set spawn point of {_existingWorld} to {maucb::spawn}
                set {mauplot::world-loaded} to true
                delete {mauplot::recovery-required}
                save {_existingWorld}
                send "&a[MAU-Generator] Die vorhandene CityBuild-Welt wurde bestätigt und wieder aktiviert." to player
                send "&7Welt und Bebauung wurden NICHT gelöscht oder neu generiert." to player
                send "&7Plots: &e10.000 &8| &7Worldborder: &e7.600 x 7.600" to player
                stop
        else:
            # Schon geladene Welt oder gespeicherter Generatorstand? Erst warnen.
            set {_existingWorld} to world named "{@mau-world-name}"
            set {_needsConfirmation} to false
            if {_existingWorld} is set:
                set {_needsConfirmation} to true
            if {mauworldgen::state} is set:
                set {_needsConfirmation} to true
            if {mauworldgen::stage} is set:
                set {_needsConfirmation} to true

            if {_needsConfirmation} is true:
                set {mauworldgen::confirm::%uuid of player%} to true
                send "&c&lACHTUNG: Eine CityBuild-Welt oder ein gespeicherter Generatorstand existiert bereits!" to player
                send "&7Es wird nichts automatisch gelöscht. Die Bestätigung lädt die vorhandene Welt und setzt sie fort bzw. verbindet sie erneut." to player
                send "&eZur Bestätigung gib ein: &a/cb welt erstellen bestaetigen" to player
                stop

        # Wenn die Welt noch nicht existiert, startet ein neuer Generatorlauf.
        # Bei bestätigtem, unvollständigem Lauf bleibt stage/progress erhalten.
        if {mauworldgen::active} is true:
            send "&c[MAU-Generator] Eine Generierung läuft bereits." to player
            stop

        set {mauworldgen::active} to true
        set {mauworldgen::state} to "running"
        set {_admin} to player

        if {mauworldgen::stage} is not set:
            set {mauworldgen::state} to "running"
            set {mauworldgen::stage} to "base"
            set {mauworldgen::progress} to -1

        send "&b[MAU-Generator] &eDie Welt wird geladen. Bitte nicht mehrfach starten." to {_admin}
        set {_w} to world named "{@mau-world-name}"
        if {_w} is not set:
            load world "{@mau-world-name}" with environment normal
            wait 2 seconds
            set {_w} to world named "{@mau-world-name}"
        if {_w} is not set:
            delete {mauworldgen::active}
            set {mauworldgen::state} to "error"
            send "&c[MAU-Generator] Die Welt konnte nicht geladen werden. Prüfe die Serverkonsole." to {_admin}
            stop

        # ----------------------------------------------------
        # PHASE 1: ZENTRALE 480x480 STARTPLATTE, CHUNKWEISE
        # 30x30 Kacheln zu je 16x16 Blöcken.
        # Pro Kachel 3 Blocklagen; nach jeder Kachel wird gewartet.
        # Fortschritt wird gespeichert und nach Neustart fortgesetzt.
        # ----------------------------------------------------
        if {mauworldgen::stage} is "base":
            send "&b[MAU-Generator] &7Phase 1/5: Grundplatte wird chunkweise gebaut ..." to {_admin}
            set {_tile} to {mauworldgen::progress} + 1
            while {_tile} <= 899:
                set {_tx} to floor({_tile} / 30)
                set {_tz} to {_tile} - ({_tx} * 30)
                set {_x1} to -240 + ({_tx} * 16)
                set {_z1} to -240 + ({_tz} * 16)
                set {_x2} to {_x1} + 15
                set {_z2} to {_z1} + 15

                set all blocks within location({_x1}, 278, {_z1}, {_w}) and location({_x2}, 278, {_z2}, {_w}) to bedrock
                set all blocks within location({_x1}, 279, {_z1}, {_w}) and location({_x2}, 279, {_z2}, {_w}) to stone
                set all blocks within location({_x1}, 280, {_z1}, {_w}) and location({_x2}, 280, {_z2}, {_w}) to gray_concrete

                set {mauworldgen::progress} to {_tile}
                add 1 to {_tile}
                wait 5 ticks
            set {mauworldgen::stage} to "plots"
            set {mauworldgen::progress} to -1

        # ----------------------------------------------------
        # PHASE 2: 36 GRASSPLOTS DER ZENTRALEN STARTSTADT
        # Jeder Plot wird in 16 Teilflächen zu je 16x16 Blöcken gebaut.
        # So werden keine kompletten 64x64-Flächen in einem Tick geändert.
        # ----------------------------------------------------
        if {mauworldgen::stage} is "plots":
            send "&b[MAU-Generator] &7Phase 2/5: zentrale 36 Startgrundstücke werden kachelweise angelegt ..." to {_admin}
            set {_tile} to {mauworldgen::progress} + 1
            while {_tile} <= 575:
                set {_plotIndex} to floor({_tile} / 16)
                set {_localTile} to {_tile} - ({_plotIndex} * 16)
                set {_plotX} to floor({_plotIndex} / 6)
                set {_plotZ} to {_plotIndex} - ({_plotX} * 6)
                set {_chunkX} to floor({_localTile} / 4)
                set {_chunkZ} to {_localTile} - ({_chunkX} * 4)
                set {_px} to -222 + ({_plotX} * 76) + ({_chunkX} * 16)
                set {_pz} to -222 + ({_plotZ} * 76) + ({_chunkZ} * 16)
                set {_px2} to {_px} + 15
                set {_pz2} to {_pz} + 15

                set all blocks within location({_px}, 279, {_pz}, {_w}) and location({_px2}, 279, {_pz2}, {_w}) to dirt
                set all blocks within location({_px}, 280, {_pz}, {_w}) and location({_px2}, 280, {_pz2}, {_w}) to grass_block

                set {mauworldgen::progress} to {_tile}
                add 1 to {_tile}
                wait 2 ticks
            set {mauworldgen::stage} to "roads"
            set {mauworldgen::progress} to -1

        # ----------------------------------------------------
        # PHASE 3: STRASSEN, GEHWEGE, MITTELLINIEN
        # ----------------------------------------------------
        if {mauworldgen::stage} is "roads":
            send "&b[MAU-Generator] &7Phase 3/5: Straßen und Gehwege ..." to {_admin}

            # Fünf innere Straßen in X- und Z-Richtung.
            loop numbers from 0 to 4:
                set {_gap} to -158 + (loop-number * 76)

                # Straße in X-Richtung: Gehwege außen, Asphalt bleibt in der Mitte.
                set all blocks within location({_gap}, 280, -240, {_w}) and location(({_gap} + 2), 280, 239, {_w}) to smooth_stone
                set all blocks within location(({_gap} + 9), 280, -240, {_w}) and location(({_gap} + 11), 280, 239, {_w}) to smooth_stone
                set all blocks within location(({_gap} + 2), 280, -240, {_w}) and location(({_gap} + 2), 280, 239, {_w}) to stone_bricks
                set all blocks within location(({_gap} + 9), 280, -240, {_w}) and location(({_gap} + 9), 280, 239, {_w}) to stone_bricks
                set all blocks within location(({_gap} + 5), 280, -240, {_w}) and location(({_gap} + 5), 280, 239, {_w}) to yellow_concrete

                # Straße in Z-Richtung.
                set all blocks within location(-240, 280, {_gap}, {_w}) and location(239, 280, ({_gap} + 2), {_w}) to smooth_stone
                set all blocks within location(-240, 280, ({_gap} + 9), {_w}) and location(239, 280, ({_gap} + 11), {_w}) to smooth_stone
                set all blocks within location(-240, 280, ({_gap} + 2), {_w}) and location(239, 280, ({_gap} + 2), {_w}) to stone_bricks
                set all blocks within location(-240, 280, ({_gap} + 9), {_w}) and location(239, 280, ({_gap} + 9), {_w}) to stone_bricks
                set all blocks within location(-240, 280, ({_gap} + 5), {_w}) and location(239, 280, ({_gap} + 5), {_w}) to yellow_concrete
                wait 1 tick

            # Äußere Gehwege und gelbe Linien entlang des Stadtrands.
            set all blocks within location(-240, 280, -240, {_w}) and location(-237, 280, 239, {_w}) to smooth_stone
            set all blocks within location(-226, 280, -240, {_w}) and location(-223, 280, 239, {_w}) to smooth_stone
            set all blocks within location(222, 280, -240, {_w}) and location(225, 280, 239, {_w}) to smooth_stone
            set all blocks within location(236, 280, -240, {_w}) and location(239, 280, 239, {_w}) to smooth_stone
            set all blocks within location(-240, 280, -240, {_w}) and location(239, 280, -237, {_w}) to smooth_stone
            set all blocks within location(-240, 280, -226, {_w}) and location(239, 280, -223, {_w}) to smooth_stone
            set all blocks within location(-240, 280, 222, {_w}) and location(239, 280, 225, {_w}) to smooth_stone
            set all blocks within location(-240, 280, 236, {_w}) and location(239, 280, 239, {_w}) to smooth_stone
            set all blocks within location(-231, 280, -240, {_w}) and location(-231, 280, 239, {_w}) to yellow_concrete
            set all blocks within location(230, 280, -240, {_w}) and location(230, 280, 239, {_w}) to yellow_concrete
            set all blocks within location(-240, 280, -231, {_w}) and location(239, 280, -231, {_w}) to yellow_concrete
            set all blocks within location(-240, 280, 230, {_w}) and location(239, 280, 230, {_w}) to yellow_concrete

            # Weiße Parkmarkierungen an den Seiten der inneren Straßen.
            loop numbers from 0 to 4:
                set {_gap} to -158 + (loop-number * 76)
                loop numbers from 0 to 5:
                    set {_segment} to -222 + (loop-number-2 * 76)
                    set {_m1} to {_segment} + 12
                    set {_m2} to {_segment} + 18
                    set {_m3} to {_segment} + 42
                    set {_m4} to {_segment} + 48

                    set all blocks within location(({_gap} + 3), 280, {_m1}, {_w}) and location(({_gap} + 4), 280, {_m1}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 7), 280, {_m1}, {_w}) and location(({_gap} + 8), 280, {_m1}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 3), 280, {_m2}, {_w}) and location(({_gap} + 4), 280, {_m2}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 7), 280, {_m2}, {_w}) and location(({_gap} + 8), 280, {_m2}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 3), 280, {_m3}, {_w}) and location(({_gap} + 4), 280, {_m3}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 7), 280, {_m3}, {_w}) and location(({_gap} + 8), 280, {_m3}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 3), 280, {_m4}, {_w}) and location(({_gap} + 4), 280, {_m4}, {_w}) to white_concrete
                    set all blocks within location(({_gap} + 7), 280, {_m4}, {_w}) and location(({_gap} + 8), 280, {_m4}, {_w}) to white_concrete
                    wait 1 tick

                    # Dieselben Parkmarkierungen in der quer verlaufenden Straße.
                    set all blocks within location({_m1}, 280, ({_gap} + 3), {_w}) and location({_m1}, 280, ({_gap} + 4), {_w}) to white_concrete
                    set all blocks within location({_m1}, 280, ({_gap} + 7), {_w}) and location({_m1}, 280, ({_gap} + 8), {_w}) to white_concrete
                    set all blocks within location({_m2}, 280, ({_gap} + 3), {_w}) and location({_m2}, 280, ({_gap} + 4), {_w}) to white_concrete
                    set all blocks within location({_m2}, 280, ({_gap} + 7), {_w}) and location({_m2}, 280, ({_gap} + 8), {_w}) to white_concrete
                    set all blocks within location({_m3}, 280, ({_gap} + 3), {_w}) and location({_m3}, 280, ({_gap} + 4), {_w}) to white_concrete
                    set all blocks within location({_m3}, 280, ({_gap} + 7), {_w}) and location({_m3}, 280, ({_gap} + 8), {_w}) to white_concrete
                    set all blocks within location({_m4}, 280, ({_gap} + 3), {_w}) and location({_m4}, 280, ({_gap} + 4), {_w}) to white_concrete
                    set all blocks within location({_m4}, 280, ({_gap} + 7), {_w}) and location({_m4}, 280, ({_gap} + 8), {_w}) to white_concrete
                    wait 1 tick

            # Spawnplatz genau in der Kreuzung in der Mitte.
            set all blocks within location(-6, 280, -6, {_w}) and location(5, 280, 5, {_w}) to polished_andesite
            set all blocks within location(-2, 280, -2, {_w}) and location(1, 280, 1, {_w}) to red_nether_bricks

            send "&aDie neue CityBuild-Welt wurde fertig erstellt!" to all players

# CITYBUILD-WELT: AUTOMATISCHES WIEDERLADEN + SPEICHERSCHUTZ
# Version 4.7
# Die CityBuild-Welt ist eine eigenstaendige Welt im Serverordner.
# Nach einem Neustart wird sie erneut geladen und mit dem Plot-System
# verbunden. Vor dem Verbinden prueft ein versteckter Layout-Fingerabdruck,
# ob wirklich die generierte Stadt geladen wurde.
# ------------------------------------------------------------

on script load:
    # Laufende Trigger koennen nach Reload/Restart nicht fortgesetzt werden.
    # Die gespeicherten Queue- und Fortschrittsvariablen bleiben erhalten.
    delete {mauworldgen::active}
    delete {mauplot::world-loaded}
    delete {mauplot::recovery-required}
    delete {mauplot::world}
    delete {maucb::citybuild-world}

    # Migration aus frueheren Versionen.
    if {mauplot::bedrock-enabled} is true:
        if {mauplot::bedrock-default-depth} is not set:
            if {mauplot::bedrock-depth} is set:
                set {mauplot::bedrock-default-depth} to {mauplot::bedrock-depth}
            else:
                set {mauplot::bedrock-default-depth} to 50

    wait 3 seconds

    if {mauworldgen::state} is "done":
        set {_savedWorld} to world named "{@mau-world-name}"

        # Eine separate Welt wird nach einem normalen Serverstart oft nicht
        # automatisch geladen. Deshalb laden wir genau diese Welt nach.
        if {_savedWorld} is not set:
            load world "{@mau-world-name}" with environment normal
            wait 3 seconds
            set {_savedWorld} to world named "{@mau-world-name}"

        if {_savedWorld} is set:
            # Layout-Fingerabdruck: Ecke der fertigen Startstadt.
            # So wird eine frisch erzeugte/leere Ersatzwelt nicht als alte
            # CityBuild-Welt ausgegeben.
            if block at location(-240, 278, -240, {_savedWorld}) is bedrock:
                if block at location(-240, 279, -240, {_savedWorld}) is stone:
                    if block at location(-240, 280, -240, {_savedWorld}) is smooth stone:
                        set {mauplot::world} to {_savedWorld}
                        set {mauplot::grid-enabled} to true
                        set {mauplot::originX} to -3794
                        set {mauplot::originZ} to -3794
                        set {mauplot::grid-step} to 76
                        set {mauplot::grid-count} to 100

                        set {_border} to world border of {_savedWorld}
                        set world border radius of {_border} to 7600
                        set world border center of {_border} to location(6, 280, 6, {_savedWorld})
                        set world border warning distance of {_border} to 25
                        set world border damage buffer of {_border} to 5
                        set worldborder of {_savedWorld} to {_border}

                        if {maucb::spawn} is not set:
                            set {maucb::spawn} to location(6, 281, 6, {_savedWorld})
                        set {maucb::citybuild-world} to {_savedWorld}
                        set spawn point of {_savedWorld} to {maucb::spawn}
                        set {mauplot::world-loaded} to true
                        delete {mauplot::recovery-required}

                        # Sofort speichern, damit Border/Spawn nach Restart
                        # auch in den Worlddaten stehen.
                        save {_savedWorld}
                        send "&a[MAU-CB] Die gespeicherte CityBuild-Welt wurde wieder geladen. Plots, Spawn und Worldborder sind verbunden." to console
                    else:
                        delete {mauplot::world-loaded}
                        set {mauplot::recovery-required} to true
                        send "&c[MAU-CB] WARNUNG: mau_citybuild_01 wurde geladen, aber der Stadt-Fingerabdruck fehlt. Spieler werden NICHT automatisch teleportiert. Bitte Weltordner/Backup pruefen; NICHT /cb welt erstellen ausfuehren!" to console
                else:
                    delete {mauplot::world-loaded}
                    set {mauplot::recovery-required} to true
                    send "&c[MAU-CB] WARNUNG: mau_citybuild_01 sieht nicht wie die gespeicherte Stadt aus. Bitte Weltordner/Backup pruefen; NICHT /cb welt erstellen ausfuehren!" to console
            else:
                delete {mauplot::world-loaded}
                set {mauplot::recovery-required} to true
                send "&c[MAU-CB] WARNUNG: Stadt-Fingerabdruck fehlt. Die Welt wird nicht als fertige CityBuild-Welt aktiviert. Bitte Backup pruefen!" to console
        else:
            delete {mauplot::world-loaded}
            set {mauplot::recovery-required} to true
            send "&c[MAU-CB] WARNUNG: mau_citybuild_01 konnte nicht geladen werden. Bitte kontrolliere den Ordner mau_citybuild_01 im Server-Dateimanager und stelle ihn ggf. aus einem Backup wieder her." to console

on join:

    if {mauplot::recovery-required} is true:

# TEIL 5/5 – CITYBUILDSET BEDROCK: 10/20/30/40/50/100 BLOECKE + GLOBALE STRASSENSTUETZEN
# /citybuildset bedrock 10 = globaler Standard für die ganze Plotwelt

# /citybuild set spawn = globalen CityBuild-Spawn setzen

command /citybuildset [<text>] [<number>]:
    executable by: players
    trigger:
        # Sämtliche Einstellungen sind Admin-Befehle. OPs oder die Admin-Permissions sind nötig.
        if mauPlotIsAdmin(player) is false:
            send "&cDafür hast du keine Berechtigung. Benötigt: &emaucb.admin &coder &emauplot.admin&c." to player
            stop

        if arg-1 is "status":
            set {_head} to {mauplot::bedrock-queue-head}
            set {_tail} to {mauplot::bedrock-queue-tail}
            if {_head} is not set:
                set {_head} to 1
            if {_tail} is not set:
                set {_tail} to 0
            set {_remaining} to {_tail} - {_head} + 1
            if {_remaining} < 0:
                set {_remaining} to 0
            set {_globalDepth} to {mauplot::bedrock-default-depth}
            if {_globalDepth} is not set:
                set {_globalDepth} to 0
            send "&8&m--------------------------------" to player
            send "&b&lCITYBUILD-UNTERGRUND STATUS" to player
            send "&7Globaler Standard: &e%{_globalDepth}% Blöcke" to player
            set {_currentKey} to mauPlotKey(location of player)
            if {_currentKey} is not "OUTSIDE":
                set {_currentDepth} to mauPlotBedrockDepth({_currentKey})
                send "&7Aktuelles Plot: &e%{_currentDepth}% Blöcke" to player
                if {_currentDepth} > 0:
                    if {mauplot::bedrock-ready-depth::%{_currentKey}%} is {_currentDepth}:
                        send "&7Plot-Status: &aFertig" to player
                    else:
                        send "&7Plot-Status: &eWird vorbereitet" to player
            send "&7Plot-Untergrund-Warteschlange: &e%{_remaining}% Regionen" to player
            set {_roadHead} to {mauplot::roadwall::head}
            set {_roadTail} to {mauplot::roadwall::tail}
            if {_roadHead} is not set:
                set {_roadHead} to 1
            if {_roadTail} is not set:
                set {_roadTail} to 0
            set {_roadRemaining} to {_roadTail} - {_roadHead} + 1
            if {_roadRemaining} < 0:
                set {_roadRemaining} to 0
            send "&7Straßen-Bruchsteinabschnitte: &e%{_roadRemaining}%" to player
            if {mauplot::bedrock-global-scan::active} is true:
                set {_scanX} to {mauplot::bedrock-global-scan::x}
                set {_scanZ} to {mauplot::bedrock-global-scan::z}
                set {_scanDone} to ({_scanX} * 100) + {_scanZ}
                send "&7Globaler Plot-/Straßen-Scan: &e%{_scanDone}%&7/&e10.000 Zellen" to player
            else:
                if {mauplot::bedrock-default-depth} is 10:
                    send "&7Globaler Scan: &aZellen-Scan abgeschlossen" to player
            if {mauplot::bedrock-queue::active} is set:
                send "&7Aktuelle Region: &e%{mauplot::bedrock-queue::active}%" to player
            send "&8&m--------------------------------" to player
            stop

        if arg-1 is not "bedrock":
            send "&cNutze: &e/citybuildset bedrock <10|20|30|40|50|100> &7oder &e/citybuildset status" to player
            stop

        if arg-2 is not set:
            send "&cBitte gib eine Tiefe an: &e10, 20, 30, 40, 50 oder 100&c." to player
            send "&7/citybuildset bedrock 10 &8- &7globaler Standard für alle Grundstücke" to player
            send "&7/citybuildset bedrock 20|30|40|50|100 &8- &7nur das Plot unter deinen Füßen" to player
            stop

        set {_depth} to arg-2
        set {_validDepth} to false
        if {_depth} is 10:
            set {_validDepth} to true
        if {_depth} is 20:
            set {_validDepth} to true
        if {_depth} is 30:
            set {_validDepth} to true
        if {_depth} is 40:
            set {_validDepth} to true
        if {_depth} is 50:
            set {_validDepth} to true
        if {_depth} is 100:
            set {_validDepth} to true
        if {_validDepth} is false:
            send "&cUngültige Tiefe. Erlaubt sind: &e10, 20, 30, 40, 50, 100&c." to player
            stop
        if {mauplot::world} is not set:
            send "&cDie Plotwelt ist noch nicht eingerichtet." to player
            stop
        if world of player is not {mauplot::world}:
            send "&cDu musst dich in der Plotwelt befinden." to player
            stop

        set {mauplot::bedrock-enabled} to true

        # 10 ist global: ein persistenter Scanner plant ALLE 10.000 Plot-Zellen ein.
        # Nur eine Zelle pro Tick wird zur Block-Warteschlange hinzugefuegt.
        if {_depth} is 10:
            set {mauplot::bedrock-default-depth} to 10
            set {mauplot::bedrock-depth} to 10
            delete {mauplot::plot-bedrock-depth::*}
            set {mauplot::bedrock-global-scan::world} to {mauplot::world}
            set {mauplot::bedrock-global-scan::x} to 0
            set {mauplot::bedrock-global-scan::z} to 0
            set {mauplot::bedrock-global-scan::queued} to 0
            set {mauplot::bedrock-global-scan::active} to true
            send "&aDer globale 10-Block-Untergrund wird fuer ALLE 10.000 Plotzellen eingeplant." to player
            send "&aAuch die Bruchstein-Stuetzen unter den seitlichen Straßenkanten werden fuer das ganze Raster eingeplant." to player
            send "&7Der Scan laeuft im Hintergrund und setzt nach Neustarts fort." to player
            send "&7Die Blockarbeiten sind sehr umfangreich und koennen viele Stunden bis Tage dauern." to player
            send "&7Fortschritt: &e/citybuildset status" to player
            stop

        # 20/30/40/50/100 verändern ausschließlich das Plot unter dem Admin.
        set {_key} to mauPlotKey(location of player)
        if {_key} is "OUTSIDE":
            send "&cStelle dich für diese Tiefe direkt auf ein Plot, nicht auf die Straße oder den Spawn." to player
            stop

        set {_parts::*} to {_key} split at "::"
        set {_plotX} to {_parts::2} parsed as number
        set {_plotZ} to {_parts::3} parsed as number
        if {_plotX} is not set:
            send "&cDas aktuelle Plot konnte nicht erkannt werden." to player
            stop
        if {_plotZ} is not set:
            send "&cDas aktuelle Plot konnte nicht erkannt werden." to player
            stop

        set {mauplot::plot-bedrock-depth::%{_key}%} to {_depth}
        mauPlotGenerateCell({_plotX}, {_plotZ}, {mauplot::world})
        set {_x1} to {mauplot::originX} + ({_plotX} * {mauplot::grid-step})
        set {_z1} to {mauplot::originZ} + ({_plotZ} * {mauplot::grid-step})
        set {_x2} to {_x1} + 63
        set {_z2} to {_z1} + 63
        mauPlotQueueBedrockRegion({_key}, {mauplot::world}, {_x1}, {_z1}, {_x2}, {_z2}, {_depth})
        set {_bedrockY} to {@mau-world-surface} - {_depth}
        send "&aDie Bautiefe dieses Plots wurde auf &e%{_depth}% Blöcke &agesetzt." to player
        send "&7Bedrock liegt bei &eY=%{_bedrockY}%&7. Andere Grundstücke bleiben unverändert." to player
        send "&7Während der Vorbereitung ist das Plot kurzzeitig gegen Bauen und Abbauen gesperrt." to player
        send "&7Fortschritt: &e/citybuildset status" to player

                set {_maxStoneLayers} to floor(({_regionDepth} + 5) / 7)

                if {mauplot::bedrock-data::phase::%{_id}%} is "floor":
                    set all blocks within location({_tileX1}, {_bedrockY}, {_tileZ1}, {_w}) and location({_tileX2}, {_bedrockY}, {_tileZ2}, {_w}) to bedrock
                    add 1 to {_tileIndex}
                    set {mauplot::bedrock-data::tile::%{_id}%} to {_tileIndex}
                    if {_tileIndex} >= {_totalTiles}:
                        set {mauplot::bedrock-data::phase::%{_id}%} to "stone"
                        set {mauplot::bedrock-data::tile::%{_id}%} to 0
                        set {mauplot::bedrock-data::layer::%{_id}%} to 0
                else if {mauplot::bedrock-data::phase::%{_id}%} is "stone":
                    set {_layer} to {mauplot::bedrock-data::layer::%{_id}%}
                    if {_layer} is not set:
                        set {_layer} to 0
                    set {_y1} to {_bedrockY} + 1 + ({_layer} * 7)
                    set {_y2} to {_y1} + 6
                    set {_surfaceBottom} to {@mau-world-surface} - 1
                    if {_y2} > {_surfaceBottom}:
                        set {_y2} to {_surfaceBottom}
                    set all blocks within location({_tileX1}, {_y1}, {_tileZ1}, {_w}) and location({_tileX2}, {_y2}, {_tileZ2}, {_w}) to stone

                    add 1 to {_layer}
                    if {_layer} >= {_maxStoneLayers}:
                        set {_layer} to 0
                        add 1 to {_tileIndex}
                        set {mauplot::bedrock-data::tile::%{_id}%} to {_tileIndex}
                    set {mauplot::bedrock-data::layer::%{_id}%} to {_layer}
                    if {_tileIndex} >= {_totalTiles}:
                        mauPlotFinishBedrockRegion({_id})
            else:
                # Welt kann nach einem Neustart noch nicht geladen sein. Queue bleibt erhalten.
                stop
